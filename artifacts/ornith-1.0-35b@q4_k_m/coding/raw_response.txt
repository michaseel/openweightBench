  <script>
    // ────────────────────────────────────────────
    // Namespace: Prevent collisions with other scripts
    // using the same localStorage key.
    // ────────────────────────────────────────────
    const APP_NAMESPACE = 'kanban-app-state';

    // ────────────────────────────────────────────
    // Column definitions (order & metadata)
    // ────────────────────────────────────────────
    const COLUMNS = ['backlog', 'in-progress', 'review', 'done'];

    // ────────────────────────────────────────────
    // Dummy data: used when no cards exist yet so
    // the board isn't empty on first load.
    // ────────────────────────────────────────────
    const DUMMY_CARDS = [
      { id: uid(), title: 'Set up project repository', column: 'backlog' },
      { id: uid(), title: 'Design wireframes',        column: 'backlog' },
      { id: uid(), title: 'Write unit tests',         column: 'in-progress' },
      { id: uid(), title: 'Implement auth flow',      column: 'in-progress' },
      { id: uid(), title: 'Code review on PR #42',   column: 'review' },
      { id: uid(), title: 'Fix responsive layout bug',column: 'done' },
    ];

    // ────────────────────────────────────────────
    // State management – load / save to localStorage
    // under our namespaced key.
    // ────────────────────────────────────────────

    /** Generate a unique ID (simple timestamp + random) */
    function uid() {
      return Date.now().toString(36) + Math.random().toString(36).slice(2, 8);
    }

    /** Load board state from localStorage; returns an object keyed by column. */
    function loadState() {
      try {
        const raw = localStorage.getItem(APP_NAMESPACE);
        if (raw) return JSON.parse(raw);
      } catch (_) { /* corrupted data – fall through to defaults */ }

      // First visit or no valid data → seed dummy cards
      const state = {};
      COLUMNS.forEach(col => { state[col] = []; });
      DUMMY_CARDS.forEach(card => {
        if (state[card.column]) state[card.column].push(card);
      });
      return state;
    }

    /** Persist current state to localStorage. */
    function saveState(state) {
      try {
        localStorage.setItem(APP_NAMESPACE, JSON.stringify(state));
      } catch (_) { /* quota exceeded – silently ignore */ }
    }

    // ────────────────────────────────────────────
    // Application state (kept in memory; synced to storage)
    // ────────────────────────────────────────────
    let boardState = loadState();

    // Track the card currently being dragged { id, column }
    let dragData = null;

    // Track which column has an open "add card" form (only one at a time)
    let activeAddColumn = null;

    // Track which card is in edit mode and its current text
    let editingCardId = null;

    // ────────────────────────────────────────────
    // DOM references – cached for performance
    // ────────────────────────────────────────────
    const boardEl        = document.getElementById('board');
    const deleteModal    = document.getElementById('delete-modal');
    const deleteMsg      = document.getElementById('delete-modal-message');

    // ─────────────────═══════════════════════════
    // RENDERING
    // ═══════════════════════════════════════════

    /** Render all columns and their cards from current state. */
    function render() {
      COLUMNS.forEach(col => {
        const container = document.querySelector(`[data-column="${col}"]`);
        if (!container) return;

        // Clear existing cards (but keep the add-card form area intact)
        container.innerHTML = '';

        boardState[col].forEach(card => {
          container.appendChild(createCardElement(card));
        });

        // Update column counter
        const countEl = document.getElementById(`count-${col}`);
        if (countEl) countEl.textContent = boardState[col].length;
      });
    }

    /** Create a single card DOM element. */
    function createCardElement(card) {
      const el = document.createElement('div');
      el.setAttribute('data-testid', 'card');
      el.setAttribute('draggable', 'true');
      el.dataset.id = card.id;
      el.className = 'kanban-card bg-white rounded-xl p-3 shadow-sm border border-gray-100 cursor-grab active:cursor-grabbing select-none';

      // Card content wrapper
      const inner = document.createElement('div');
      inner.className = 'flex items-start justify-between gap-2 group/card';

      // Title (click to edit in place)
      const titleEl = document.createElement('span');
      titleEl.className = 'text-sm text-gray-700 font-medium leading-snug flex-1 break-words card-title cursor-text';
      titleEl.textContent = card.title;
      titleEl.addEventListener('dblclick', () => enterEditMode(card.id, titleEl));

      // Delete button (visible on hover via group)
      const deleteBtn = document.createElement('button');
      deleteBtn.setAttribute('data-testid', 'delete-card');
      deleteBtn.className = 'opacity-0 group-hover/card:opacity-100 transition-opacity p-1 text-gray-400 hover:text-red-500 rounded';
      deleteBtn.title = 'Delete card';
      deleteBtn.innerHTML = '<svg class="w-3.5 h-3.5" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M6 18L18 6M6 6l12 12"/></svg>';
      deleteBtn.addEventListener('click', (e) => {
        e.stopPropagation();
        showDeleteModal(card.id, card.title);
      });

      inner.appendChild(titleEl);
      inner.appendChild(deleteBtn);
      el.appendChild(inner);

      // ── Drag events on the card element ──
      el.addEventListener('dragstart', (e) => {
        dragData = { id: card.id, column: findCardColumn(card.id) };
        el.classList.add('card-dragging');
        e.dataTransfer.effectAllowed = 'move';
        // Set a transparent drag image for a cleaner look
        const ghost = el.cloneNode(true);
        ghost.style.width = `${el.offsetWidth}px`;
        ghost.classList.remove('card-dragging');
        ghost.style.opacity = '0.6';
        e.dataTransfer.setDragImage(ghost, 20, 20);
      });

      el.addEventListener('dragend', () => {
        el.classList.remove('card-dragging');
        dragData = null;
        // Remove all drop highlights
        document.querySelectorAll('.column-drop-highlight').forEach(el => el.classList.remove('column-drop-highlight'));
      });

      return el;
    }

    /** Find which column an card ID belongs to. */
    function findCardColumn(cardId) {
      for (const col of COLUMNS) {
        if (boardState[col].some(c => c.id === cardId)) return col;
      }
      return null;
    }

    // ─────────────────═══════════════════════════
    // DRAG AND DROP – Column-level handlers
    // ═══════════════════════════════════════════

    /** Attach drag-over / drop listeners to every cards-container. */
    function initDragDrop() {
      document.querySelectorAll('.cards-container').forEach(container => {
        const col = container.dataset.column;

        container.addEventListener('dragover', (e) => {
          e.preventDefault();               // Allow dropping
          e.dataTransfer.dropEffect = 'move';
          container.closest('[data-testid]').classList.add('column-drop-highlight');
        });

        container.addEventListener('dragleave', (e) => {
          // Only remove highlight if we actually left the column area
          if (!container.contains(e.relatedTarget)) {
            container.closest('[data-testid]').classList.remove('column-drop-highlight');
          }
        });

        container.addEventListener('drop', (e) => {
          e.preventDefault();
          container.closest('[data-testid]').classList.remove('column-drop-highlight');

          if (!dragData) return;

          const sourceCol = dragData.column;
          const destCol   = col;
          const cardId    = dragData.id;

          // Remove card from source column
          boardState[sourceCol] = boardState[sourceCol].filter(c => c.id !== cardId);

          // Add to destination column
          boardState[destCol].push({ id: cardId, title: findCardTitle(cardId) });

          saveState(boardState);
          render();

          // Trigger confetti if dropped into "Done" (and it wasn't already there)
          if (destCol === 'done' && sourceCol !== 'done') {
            triggerConfetti();
          }
        });
      });
    }

    /** Look up a card's title from current state. */
    function findCardTitle(cardId) {
      for (const col of COLUMNS) {
        const found = boardState[col].find(c => c.id === cardId);
        if (found) return found.title;
      }
      return 'Untitled';
    }

    // ─────────────────═══════════════════════════
    // ADD CARD – Inline form per column
    // ═══════════════════════════════════════════

    /** Wire up all "Add card" buttons. */
    function initAddCard() {
      document.querySelectorAll('[data-testid="add-card"]').forEach(btn => {
        btn.addEventListener('click', () => openAddForm(btn.dataset.column));
      });
    }

    /** Show the inline input form for a given column; hide others. */
    function openAddForm(column) {
      // Close any previously open form
      closeAllForms();

      activeAddColumn = column;
      const section   = document.querySelector(`[data-column="${column}"]`).closest('[data-testid]');
      const form      = section.querySelector('.add-card-form');
      const input     = section.querySelector('[data-testid="card-input"]');

      if (!form || !input) return;

      form.classList.remove('hidden');
      input.value = '';
      setTimeout(() => input.focus(), 50);

      // Enter commits the card
      input.addEventListener('keydown', (e) => {
        if (e.key === 'Enter' && !e.shiftKey) {
          e.preventDefault();
          commitNewCard(column, input.value.trim());
        }
        if (e.key === 'Escape') {
          closeAllForms();
        }
      });

      // Confirm button
      const confirmBtn = section.querySelector('.add-card-confirm');
      confirmBtn.onclick = () => commitNewCard(column, input.value.trim());

      // Cancel button
      const cancelBtn = section.querySelector('.add-card-cancel');
      cancelBtn.onclick = closeAllForms;
    }

    /** Create a new card in the given column. */
    function commitNewCard(column, title) {
      if (!title || !title.trim()) return;           // Ignore empty titles
      const trimmed = title.trim();

      boardState[column].push({ id: uid(), title: trimmed });
      saveState(boardState);
      render();
      closeAllForms();
    }

    /** Hide all inline add-card forms. */
    function closeAllForms() {
      document.querySelectorAll('.add-card-form').forEach(f => f.classList.add('hidden'));
      activeAddColumn = null;
    }

    // ─────────────────═══════════════════════════
    // EDIT IN PLACE – Double-click to edit title
    // ═══════════════════════════════════════════

    /** Replace a card's title span with an editable textarea. */
    function enterEditMode(cardId, titleEl) {
      const col = findCardColumn(cardId);
      if (!col) return;
      const card = boardState[col].find(c => c.id === cardId);
      if (!card) return;

      editingCardId = cardId;
      const parent  = titleEl.parentElement;
      const textarea = document.createElement('textarea');
      textarea.className = 'w-full px-2 py-1 rounded-lg border border-indigo-300 text-sm text-gray-700 font-medium resize-none card-input leading-snug';
      textarea.value = card.title;
      textarea.rows = 2;

      // Commit on Enter (no shift), cancel on Escape
      textarea.addEventListener('keydown', (e) => {
        if (e.key === 'Enter' && !e.shiftKey) {
          e.preventDefault();
          exitEditMode(cardId, col, card, textarea);
        }
        if (e.key === 'Escape') {
          editingCardId = null;
          render(); // Re-render to discard changes
        }
      });

      textarea.addEventListener('blur', () => {
        // Small delay so the Enter keyup doesn't trigger a premature blur
        setTimeout(() => {
          if (editingCardId === cardId) exitEditMode(cardId, col, card, textarea);
        }, 150);
      });

      parent.replaceChild(textarea, titleEl);
      textarea.focus();
      textarea.select();
    }

    /** Save edit and restore the title span. */
    function exitEditMode(cardId, column, card, textarea) {
      const newTitle = textarea.value.trim();
      if (newTitle) {
        card.title = newTitle;
        saveState(boardState);
      }
      editingCardId = null;
      render();
    }

    // ─────────────────═══════════════════════════
    // DELETE – Custom modal (no window.alert/prompt)
    // ═══════════════════════════════════════════

    /** Show the confirmation dialog for deleting a card. */
    function showDeleteModal(cardId, title) {
      deleteMsg.textContent = `"${title}" will be permanently removed.`;
      deleteModal.classList.remove('hidden');
      deleteModal.classList.add('flex');

      // Wire confirm button (once is enough; we re-query each time for safety)
      const confirmBtn = deleteModal.querySelector('[data-testid="confirm-delete"]');
      confirmBtn.onclick = () => {
        deleteCard(cardId);
        closeDeleteModal();
      };

      const cancelBtn = deleteModal.querySelector('[data-testid="cancel-delete"]');
      cancelBtn.onclick = closeDeleteModal;

      // Close on backdrop click
      deleteModal.addEventListener('click', (e) => {
        if (e.target === deleteModal || e.target.classList.contains('modal-overlay')) {
          closeDeleteModal();
        }
      });

      // Focus the confirm button for accessibility
      confirmBtn.focus();
    }

    /** Close the delete confirmation modal. */
    function closeDeleteModal() {
      deleteModal.classList.add('hidden');
      deleteModal.classList.remove('flex');
    }

    /** Remove a card from its column and persist. */
    function deleteCard(cardId) {
      for (const col of COLUMNS) {
        const idx = boardState[col].findIndex(c => c.id === cardId);
        if (idx !== -1) {
          boardState[col].splice(idx, 1);
          saveState(boardState);
          render();
          return;
        }
      }
    }

    // ─────────────────═══════════════════════════
    // CONFETTI – Lightweight particle animation
    // ═══════════════════════════════════════════

    /** Spawn a burst of colorful confetti particles across the screen. */
    function triggerConfetti() {
      const colors = [
        '#f43f5e', '#ec4899', '#a855f7', '#8b5cf6',
        '#6366f1', '#3b82f6', '#10b981', '#f59e0b',
        '#ef4444', '#14b8a6',
      ];

      const particleCount = 60;

      for (let i = 0; i < particleCount; i++) {
        const piece = document.createElement('div');
        piece.className = 'confetti-piece';

        // Randomize appearance and trajectory
        const color       = colors[Math.floor(Math.random() * colors.length)];
        const size         = Math.random() * 8 + 6;          // 6–14px
        const left         = Math.random() * 100;            // 0–100vw
        const duration     = Math.random() * 2 + 1.5;       // 1.5–3.5s
        const delay        = Math.random() * 0.6;           // 0–0.6s stagger
        const shape        = Math.random() > 0.5 ? '50%' : '2px'; // circle or square

        piece.style.cssText = `
          background: ${color};
          width:  ${size}px;
          height: ${size}px;
          border-radius: ${shape};
          left: ${left}vw;
          top: -20px;
          animation-duration: ${duration}s;
          animation-delay: ${delay}s;
        `;

        document.body.appendChild(piece);

        // Remove from DOM after animation completes
        setTimeout(() => piece.remove(), (duration + delay) * 1000 + 200);
      }
    }

    // ─────────────────═══════════════════════════
    // RESET – Wipe localStorage & re-seed dummies
    // ═══════════════════════════════════════════

    function initReset() {
      const resetBtn = document.getElementById('reset-board');
      if (resetBtn) {
        resetBtn.addEventListener('click', () => {
          localStorage.removeItem(APP_NAMESPACE);
          boardState = loadState();
          render();
        });
      }
    }

    // ─────────────────═══════════════════════════
    // INITIALIZATION – Run on DOM ready

    function init() {
      render();
      initDragDrop();
      initAddCard();
      initReset();
    }

    // Wait for the DOM to be fully parsed before wiring everything up.
    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', init);
    } else {
      init();
    }
