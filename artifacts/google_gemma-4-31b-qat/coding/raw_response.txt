    <script>
        /**
         * APP STATE & CONSTANTS
         */
        const STORAGE_KEY = 'vibrant_kanban_board_state_v1';
        let state = {
            cards: []
        };
        let cardToDeleteId = null;

        // Initial dummy data
        const DUMMY_CARDS = [
            { id: '1', text: 'Research competitors', status: 'column-backlog' },
            { id: '2', text: 'Design system architecture', status: 'column-backlog' },
            { id: '3', text: 'Implement Drag & Drop', status: 'column-in-progress' },
            { id: '4', text: 'Write unit tests', status: 'column-review' },
            { id: '5', text: 'Project Setup', status: 'column-done' },
        ];

        /**
         * CORE FUNCTIONS
         */

        // Load state from localStorage or use dummy data
        function init() {
            const saved = localStorage.getItem(STORAGE_KEY);
            if (saved) {
                state = JSON.parse(saved);
            } else {
                state.cards = [...DUMMY_CARDS];
                saveState();
            }
            renderBoard();
        }

        function saveState() {
            localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
        }

        // Renders the entire board based on state
        function renderBoard() {
            const columns = ['column-backlog', 'column-in-progress', 'column-review', 'column-done'];
            
            columns.forEach(colId => {
                const container = document.querySelector(`[data-testid="${colId}"]`);
                container.innerHTML = ''; // Clear current cards
                
                const columnCards = state.cards.filter(c => c.status === colId);
                
                columnCards.forEach(card => {
                    container.appendChild(createCardElement(card));
                });
            });
        }

        // Creates a card DOM element
        function createCardElement(card) {
            const cardEl = document.createElement('div');
            cardEl.setAttribute('data-testid', 'card');
            cardEl.setAttribute('draggable', 'true');
            cardEl.setAttribute('id', `card-${card.id}`);
            cardEl.className = 'card-transition group bg-white p-4 rounded-xl shadow-sm border border-slate-200 mb-3 cursor-grab active:cursor-grabbing hover:shadow-md hover:border-indigo-300 relative';
            
            // Drag events
            cardEl.addEventListener('dragstart', (e) => {
                e.dataTransfer.setData('text/plain', card.id);
                cardEl.classList.add('dragging');
            });
            cardEl.addEventListener('dragend', () => {
                cardEl.classList.remove('dragging');
            });

            // Text content wrapper
            const textSpan = document.createElement('span');
            textSpan.className = 'block pr-6 text-slate-700 font-medium outline-none focus:text-indigo-600';
            textSpan.textContent = card.text;
            
            // In-place editing logic
            textSpan.addEventListener('dblclick', () => {
                const input = document.createElement('input');
                input.className = 'w-full p-1 border border-indigo-400 rounded outline-none text-slate-700 font-medium';
                input.value = card.text;
                input.focus();
                
                const handleSave = () => {
                    const newText = input.value.trim();
                    if (newText) {
                        updateCardText(card.id, newText);
                    } else {
                        renderBoard(); // Revert if empty
                    }
                };

                input.addEventListener('blur', handleSave);
                input.addEventListener('keydown', (e) => {
                    if (e.key === 'Enter') handleSave();
                    if (e.key === 'Escape') renderBoard();
                });

                cardEl.replaceChild(input, textSpan);
            });

            // Delete button
            const delBtn = document.createElement('button');
            delBtn.setAttribute('data-testid', 'delete-card');
            delBtn.innerHTML = `<svg xmlns="http://www.w3.org/2000/svg" class="h-4 w-4" viewBox="0 0 20 20" fill="currentColor"><path fill-rule="evenodd" d="M4.293 4.293a1 1 0 011.414 0L10 8.586l4.293-4.293a1 1 0 111.414 1.414L11.414 10l4.293 4.293a1 1 0 01-1.414 1.414L10 11.414l-4.293 4.293a1 1 0 01-1.414-1.414L8.586 10 4.293 5.707a1 1 0 010-1.414z" clip-rule="evenodd" /></svg>`;
            delBtn.className = 'absolute top-3 right-3 text-slate-300 hover:text-red-500 opacity-0 group-hover:opacity-100 transition-opacity';
            delBtn.onclick = () => openConfirmModal(card.id);

            cardEl.appendChild(textSpan);
            cardEl.appendChild(delBtn);
            
            return cardEl;
        }

        /**
         * ACTION HANDLERS
         */

        function showAddInput(columnId) {
            const container = document.querySelector(`[data-testid="${columnId}"]`);
            const btn = container.nextElementSibling; // The "Add Card" button
            
            // Create input field
            const inputWrapper = document.createElement('div');
            inputWrapper.className = 'flex gap-2 mb-4 animate-in fade-in slide-in-from-top-1 duration-200';
            
            const input = document.createElement('input');
            input.setAttribute('data-testid', 'card-input');
            input.className = 'flex-1 p-2 rounded-lg border-2 border-indigo-500 outline-none text-slate-700';
            input.placeholder = 'What needs to be done?';
            
            const addBtn = document.createElement('button');
            addBtn.className = 'bg-indigo-600 text-white px-4 py-2 rounded-lg hover:bg-indigo-700 transition-colors font-bold';
            addBtn.textContent = 'Add';
            
            const cancelBtn = document.createElement('button');
            cancelBtn.className = 'bg-slate-200 text-slate-600 px-4 py-2 rounded-lg hover:bg-slate-300 transition-colors font-bold';
            cancelBtn.textContent = 'Cancel';

            const commit = () => {
                const val = input.value.trim();
                if (val) {
                    const newCard = {
                        id: Date.now().toString(),
                        text: val,
                        status: columnId
                    };
                    state.cards.push(newCard);
                    saveState();
                    renderBoard();
                } else {
                    container.innerHTML = ''; // Simple cleanup if empty
                }
                // Restore button
                btn.classList.remove('hidden');
                document.body.removeChild(inputWrapper); // We'll actually append to container
            };

            // Since we want the input to be part of the column visually
            inputWrapper.appendChild(input);
            inputWrapper.appendChild(addBtn);
            inputWrapper.appendChild(cancelBtn);
            
            // Prepend to container
            container.prepend(inputWrapper);
            btn.classList.add('hidden');
            input.focus();

            addBtn.onclick = commit;
            cancelBtn.onclick = () => {
                btn.classList.remove('hidden');
                container.removeChild(inputWrapper);
            };
            input.onkeydown = (e) => {
                if (e.key === 'Enter') commit();
                if (e.key === 'Escape') {
                    btn.classList.remove('hidden');
                    container.removeChild(inputWrapper);
                }
            };
        }

        function updateCardText(id, text) {
            const card = state.cards.find(c => c.id === id);
            if (card) {
                card.text = text;
                saveState();
                renderBoard();
            }
        }

        /**
         * MODAL LOGIC
         */
        function openConfirmModal(id) {
            cardToDeleteId = id;
            const modal = document.getElementById('confirm-modal');
            modal.classList.remove('hidden');
            setTimeout(() => {
                document.getElementById('modal-content').classList.remove('scale-95');
                document.getElementById('modal-content').classList.add('scale-100');
            }, 10);
        }

        function closeConfirmModal() {
            cardToDeleteId = null;
            const modal = document.getElementById('confirm-modal');
            document.getElementById('modal-content').classList.add('scale-95');
            document.getElementById('modal-content').classList.remove('scale-100');
            setTimeout(() => {
                modal.classList.add('hidden');
            }, 200);
        }

        function executeDelete() {
            if (cardToDeleteId) {
                state.cards = state.cards.filter(c => c.id !== cardToDeleteId);
                saveState();
                renderBoard();
            }
            closeConfirmModal();
        }

        /**
         * DRAG AND DROP API
         */
        function allowDrop(e) {
            e.preventDefault();
            const container = e.currentTarget;
            container.classList.add('drop-zone-active');
        }

        function clearDropZone(e) {
            e.currentTarget.classList.remove('drop-zone-active');
        }

        function drop(e) {
            e.preventDefault();
            const container = e.currentTarget;
            container.classList.remove('drop-zone-active');
            
            const cardId = e.dataTransfer.getData('text/plain');
            const newStatus = container.getAttribute('data-testid');
            
            const card = state.cards.find(c => c.id === cardId);
            if (card) {
                const oldStatus = card.status;
                card.status = newStatus;
                saveState();
                renderBoard();
                
                // Celebration if moved to Done
                if (newStatus === 'column-done' && oldStatus !== 'column-done') {
                    triggerConfetti();
                }
            }
        }

        function triggerConfetti() {
            confetti({
                particleCount: 100,
                spread: 70,
                origin: { y: 0.6 },
                colors: ['#6366f1', '#8b5cf6', '#10b981']
            });
        }

        // Start the app
        init();
    </script>
</body>
