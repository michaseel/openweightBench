    <script>
        document.addEventListener('DOMContentLoaded', () => {
            // Configuration
            const STORAGE_NAMESPACE = 'kanban-board';
            const COLUMNS = {
                'backlog': data-testid="column-backlog",
                'in-progress': data-testid="column-in-progress",
                'review': data-testid="column-review",
                'done': data-testid="column-done"
            };
            
            // Initialize state
            let boardState = {
                backlog: [],
                'in-progress': [],
                review: [],
                done: []
            };
            
            // Load from localStorage or initialize with dummy cards
            const loadState = () => {
                const saved = localStorage.getItem(STORAGE_NAMESPACE);
                if (saved) {
                    try {
                        boardState = JSON.parse(saved);
                        return true;
                    } catch (e) {
                        console.error("Failed to parse saved state", e);
                    }
                }
                return false;
            };
            
            // Save to localStorage
            const saveState = () => {
                localStorage.setItem(STORAGE_NAMESPACE, JSON.stringify(boardState));
            };
            
            // Initialize with dummy cards if needed
            const initializeDummyCards = () => {
                const allColumns = Object.keys(COLUMNS);
                allColumns.forEach(column => {
                    if (boardState[column].length === 0) {
                        for (let i = 1; i <= 3; i++) {
                            boardState[column].push({
                                id: `${Date.now()}-${Math.random().toString(36).substr(2, 5)}`,
                                title: `Dummy Card ${i} in ${column.replace('-', ' ')}`
                            });
                        }
                    }
                });
            };
            
            // Render the board
            const renderBoard = () => {
                Object.keys(COLUMNS).forEach(column => {
                    const container = document.querySelector(COLUMNS[column]);
                    container.innerHTML = '';
                    
                    boardState[column].forEach(card => {
                        const cardElement = document.createElement('div');
                        cardElement.className = 'card bg-white rounded-xl shadow-md mb-4 transition-all duration-200 hover:shadow-xl cursor-grab';
                        cardElement.dataset.id = card.id;
                        cardElement.innerHTML = `
                            <div class="p-4">
                                <div class="flex justify-between items-start">
                                    <span class="text-gray-800 font-medium line-clamp-2">${card.title}</span>
                                    <button class="delete-btn text-red-500 hover:text-red-700 ml-2" data-testid="delete-card">
                                        <svg xmlns="http://www.w3.org/2000/svg" className="h-5 w-5" viewBox="0 0 20 20" fill="currentColor">
                                            <path fillRule="evenodd" d="M4.293 4.293a1 1 0 011.414 0L10 8.586l4.293-4.293a1 1 0 111.414 1.414L11.414 10l4.293 4.293a1 1 0 01-1.414 1.414L10 11.414l-4.293 4.293a1 1 0 01-1.414-1.414L8.586 10 4.293 5.707a1 1 0 010-1.414z" clipRule="evenodd" />
                                        </svg>
                                    </button>
                                </div>
                            </div>
                        `;
                        container.appendChild(cardElement);
                    });
                });
            };
            
            // Add new card
            const addCard = (column) => {
                const input = document.createElement('input');
                input.type = "text";
                input.className = "w-full p-2 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-blue-500 mt-2";
                input.placeholder = "Enter card title...";
                input.dataset.testid = "card-input";
                
                const addButton = document.querySelector(`[data-testid="add-card"]`);
                const container = document.querySelector(COLUMNS[column]);
                const addCardContainer = container.querySelector('.add-card-container');
                
                // Clear existing content and show input
                container.innerHTML = '';
                container.appendChild(input);
                input.focus();
                
                const handleAdd = (title) => {
                    if (title.trim() === '') return;
                    
                    const newCard = {
                        id: `${Date.now()}-${Math.random().toString(36).substr(2, 5)}`,
                        title: title.trim()
                    };
                    
                    boardState[column].push(newCard);
                    saveState();
                    renderBoard();
                    
                    // Hide input and show add button again
                    container.innerHTML = '';
                    container.appendChild(addButton);
                };
                
                input.addEventListener('keypress', (e) => {
                    if (e.key === 'Enter') {
                        handleAdd(input.value);
                    }
                });
                
                // Cancel on Escape
                input.addEventListener('keydown', (e) => {
                    if (e.key === 'Escape') {
                        container.innerHTML = '';
                        container.appendChild(addButton);
                    }
                });
            };
            
            // Edit card
            const editCard = (cardElement, titleSpan) => {
                const input = document.createElement('input');
                input.type = "text";
                input.className = "edit-input";
                input.value = titleSpan.textContent;
                input.focus();
                
                titleSpan.parentNode.replaceChild(input, titleSpan);
                
                const handleSave = () => {
                    if (input.value.trim() !== '') {
                        boardState[cardElement.dataset.id] = {
                            ...boardState[cardElement.dataset.id],
                            title: input.value.trim()
                        };
                        saveState();
                    }
                    titleSpan.textContent = input.value;
                    titleSpan.parentNode.replaceChild(titleSpan, input);
                };
                
                const handleCancel = () => {
                    titleSpan.textContent = input.value;
                    titleSpan. parentNode. replaceChild(titleSpan, input);
                };
                
                input.addEventListener('keypress', (e) => {
                    if (e.key === 'Enter') handleSave();
                    if (e.key === 'Escape') handleCancel();
                });
                
                window.addEventListener('click', (e) => {
                    if (!e.target.closest('.edit-input')) {
                        handleCancel();
                    }
                });
            };
            
            // Delete card with confirmation
            let cardToDelete = null;
            
            const showDeleteConfirmation = (cardElement, title) => {
                document.getElementById('modal-title').textContent = `Delete "${title}"?`;
                document.getElementById('modal-message').textContent = "Are you sure you want to delete this card? This action cannot be undone.";
                cardToDelete = cardElement.dataset.id;
                document.getElementById('delete-modal').classList.remove('hidden');
            };
            
            const confirmDelete = () => {
                if (cardToDelete) {
                    boardState[Object.keys(COLUMNS).find(col => 
                        boardState[col].some(card => card.id === cardToDelete)
                    )] = boardState[Object.keys(COLUMNS).find(col => 
                        boardState[col].some(card => card.id === cardToDelete)
                    )].filter(card => card.id !== cardToDelete);
                    
                    saveState();
                    renderBoard();
                    document.getElementById('delete-modal').classList.add('hidden');
                    cardToDelete = null;
                }
            };
            
            // Trigger confetti animation
            const triggerConfetti = () => {
                const doneColumn = document.querySelector(COLUMNS.done);
                if (!doneColumn) return;
                
                const particles = [];
                const particleCount = 15;
                
                for (let i = 0; i < particleCount; i++) {
                    const particle = document.createElement('div');
                    particle.className = 'confetti-piece';
                    particle.style.left = `${Math.random() * 100}%`;
                    particle.style.backgroundColor = `hsl(${Math.random() * 360}, 70%, 60%)`;
                    particle.style.width = `${Math.random() * 10 + 5}px`;
                    particle.style.height = `${Math.random() * 10 + 5}px`;
                    doneColumn.appendChild(particle);
                    particles.push(particle);
                }
                
                setTimeout(() => {
                    particles.forEach(particle => particle.remove());
                }, 2000);
            };
            
            // Drag and Drop Setup
            let draggedCard = null;
            
            document.addEventListener('dragstart', (e) => {
                if (e.target.classList.contains('card')) {
                    draggedCard = e.target;
                    e.dataTransfer.effectAllowed = 'move';
                    e.dataTransfer.setData('text/html', e.target.outerHTML);
                    setTimeout(() => e.target.remove(), 0);
                }
            });
            
            document.addEventListener('dragover', (e) => {
                if (e.target.classList.contains('card-container')) {
                    e.preventDefault();
                    e.dataTransfer.dropEffect = 'move';
                    e.target.classList.add('drag-over');
                }
            });
            
            document.addEventListener('dragleave', (e) => {
                if (e.target.classList.contains('card-container')) {
                    e.target.classList.remove('drag-over');
                }
            });
            
            document.addEventListener('drop', (e) => {
                if (e.target.classList. contains ('card-container')) {
                    e.preventDefault();
                    e.target.classList.remove('drag-over');
                    
                    const targetColumn = Object.keys(COLUMNS).find(col => 
                        document.querySelector(COLUMNS[col]) === e.target
                    );
                    
                    if (targetColumn && draggedCard) {
                        const newCard = {
                            id: `${Date.now()}-${Math.random().toString(36).substr(2, 5)}`,
                            title: draggedCard.querySelector('.line-clamp-2').textContent
                        };
                        
                        boardState[targetColumn].push(newCard);
                        saveState();
                        renderBoard();
                        
                        if (targetColumn === 'done') {
                            triggerConfetti();
                        }
                    }
                }
            });
            
            // Initialize and Render
            if (!loadState()) {
                initializeDummyCards();
                saveState();
            }
            
            renderBoard();
            
            // Event Listeners
            Object.keys(COLUMNS).forEach(column => {
                const container = document.querySelector(COLUMNS[column]);
                const addButton = container.querySelector('[data-testid="add-card"]');
                
                addButton.addEventListener('click', () => addCard(column));
                
                // Delete confirmation modal
                document.getElementById('cancel-delete').addEventListener('click', () => {
                    document.getElementById('delete-modal').classList.add('hidden');
                    cardToDelete = null;
                });
                
                document.getElementById('confirm-delete').addEventListener('click', confirmDelete);
            });
            
            // Global event delegation for card interactions
            document.addEventListener('click', (e) => {
                if (e.target.closest('[data-testid="delete-card"]')) {
                    const cardElement = e.target.closest('.card');
                    const titleSpan = cardElement.querySelector('.line-clamp-2');
                    showDeleteConfirmation(cardElement, titleSpan.textContent);
                }
            });
            
            document.addEventListener('dblclick', (e) => {
                if (e.target.closest('.p-4')) {
                    const titleSpan = e.target.closest('.line-clamp-2');
                    editCard(e.target.closest('.card'), titleSpan);
                }
            });
        });
    </script>
